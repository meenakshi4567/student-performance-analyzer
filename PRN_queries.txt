-- ── LOGIN SYSTEM ──────────────────────────────────────────────

-- Fetch an active user by username to verify login credentials at the DB level
SELECT * FROM users WHERE username = %s AND is_active = 1;

-- Record every login attempt (success or failure) in the audit trail
INSERT INTO audit_log (user_id, action, details) VALUES (%s, %s, %s);


-- ── STUDENT CRUD ──────────────────────────────────────────────

-- Insert a new student record supplied by admin or faculty at runtime
INSERT INTO students (name, department, marks) VALUES (%s, %s, %s);

-- Insert a new student and link them to their user account (used during create_user)
INSERT INTO students (name, department, marks, user_id) VALUES (%s, %s, %s, %s);

-- Retrieve all students sorted by ID for the full student list view
SELECT id, name, department, marks, enrolled_at FROM students ORDER BY id;

-- Search students by department; LOWER() ensures case-insensitive matching
SELECT id, name, department, marks
FROM students
WHERE LOWER(department) = LOWER(%s)
ORDER BY marks DESC;

-- Update a student's name, department, and marks identified by their ID
UPDATE students SET name = %s, department = %s, marks = %s WHERE id = %s;

-- Hard-delete a student row by ID (Admin only; confirmation required in Python)
DELETE FROM students WHERE id = %s;

-- Fetch only the record belonging to the currently logged-in student user
SELECT * FROM students WHERE user_id = %s;


-- ── ANALYTICS DASHBOARD ───────────────────────────────────────

-- Compute overall stats in one pass: count, average, highest, lowest marks
SELECT COUNT(*) AS total,
       ROUND(AVG(marks), 2) AS average,
       MAX(marks) AS highest,
       MIN(marks) AS lowest
FROM students;

-- Count students per grade band using CASE inside SUM to avoid multiple queries
SELECT
    SUM(CASE WHEN marks >= 90                THEN 1 ELSE 0 END) AS grade_A,
    SUM(CASE WHEN marks >= 75 AND marks < 90 THEN 1 ELSE 0 END) AS grade_B,
    SUM(CASE WHEN marks >= 60 AND marks < 75 THEN 1 ELSE 0 END) AS grade_C,
    SUM(CASE WHEN marks >= 40 AND marks < 60 THEN 1 ELSE 0 END) AS grade_D,
    SUM(CASE WHEN marks < 40                 THEN 1 ELSE 0 END) AS grade_F
FROM students;

-- Aggregate per-department stats with GROUP BY for the department summary panel
SELECT department,
       COUNT(*)            AS total_students,
       ROUND(AVG(marks),2) AS avg_marks,
       MAX(marks)          AS highest_marks,
       MIN(marks)          AS lowest_marks
FROM students
GROUP BY department
ORDER BY avg_marks DESC;


-- ── TOP PERFORMERS & LEADERBOARD ──────────────────────────────

-- Return the three students with the highest marks using ORDER BY + LIMIT
SELECT name, department, marks
FROM students
ORDER BY marks DESC
LIMIT 3;

-- Rank every department by average marks descending for the leaderboard
SELECT department,
       ROUND(AVG(marks), 2) AS avg_marks,
       COUNT(*)             AS students
FROM students
GROUP BY department
ORDER BY avg_marks DESC;


-- ── BONUS MARKS ───────────────────────────────────────────────

-- Add bonus marks to all students in a department; LEAST caps the result at 100
UPDATE students
SET marks = LEAST(marks + %s, 100)
WHERE LOWER(department) = LOWER(%s);


-- ── EXPORT HIGH PERFORMERS ────────────────────────────────────

-- Select students whose marks exceed the overall average (subquery) for CSV export
SELECT name, department, marks
FROM students
WHERE marks > (SELECT AVG(marks) FROM students)
ORDER BY marks DESC;


-- ── USER MANAGEMENT ───────────────────────────────────────────

-- Insert a new application user with a hashed password and chosen role
INSERT INTO users (username, password_hash, role) VALUES (%s, %s, %s);

-- List all users for admin display; password_hash is intentionally excluded
SELECT id, username, role, is_active, created_at
FROM users
ORDER BY id;

-- Flip a user's active flag without knowing the current value (toggle pattern)
UPDATE users SET is_active = NOT is_active WHERE id = %s;


-- ── AUDIT LOG ─────────────────────────────────────────────────

-- Retrieve the 20 most recent audit entries sorted newest-first for admin review
SELECT id, user_id, action, details, timestamp
FROM audit_log
ORDER BY timestamp DESC
LIMIT 20;


-- ── MYSQL APPLICATION USER GRANTS ─────────────────────────────

-- Create spa_admin with full database access to mirror the admin role
CREATE USER IF NOT EXISTS 'spa_admin'@'localhost' IDENTIFIED BY 'admin_pass';
GRANT ALL PRIVILEGES ON spa_db.* TO 'spa_admin'@'localhost';

-- Create spa_faculty with read/write on students and read-only on users
CREATE USER IF NOT EXISTS 'spa_faculty'@'localhost' IDENTIFIED BY 'faculty_pass';
GRANT SELECT, INSERT, UPDATE ON spa_db.students  TO 'spa_faculty'@'localhost';
GRANT SELECT                  ON spa_db.users     TO 'spa_faculty'@'localhost';
GRANT INSERT                  ON spa_db.audit_log TO 'spa_faculty'@'localhost';

-- Create spa_student with read-only access matching the student role restrictions
CREATE USER IF NOT EXISTS 'spa_student'@'localhost' IDENTIFIED BY 'student_pass';
GRANT SELECT ON spa_db.students TO 'spa_student'@'localhost';
GRANT SELECT ON spa_db.users    TO 'spa_student'@'localhost';

-- Apply all grant changes immediately
FLUSH PRIVILEGES;
